    @Getter
    @AllArgsConstructor
    @Builder
    public class Experiment {
        private ExperimentId id; // identity
        private ExperimentDescription description; // properties package
        private ExperimentGroup group; // ExperimentGroup Object
        private ExperimentBuckets buckets; // collection of associated Bucket

        public record ExperimentId(String experimentId){}

        @Builder
        public record ExperimentDescription(String name, String description, String condition, double percentage, LocalDateTime createdAt, LocalDateTime updatedAt, ExperimentStatus staticStatus, DynamicStatus dynamicStatus, String featureKey, LocalDateTime startTime, LocalDateTime endTime){
            public enum ExperimentStatus {
                DRAFT, PUBLISHED, PAUSED, STOPPED
            }
            public enum DynamicStatus {
                DRAFT, NOT_BEGIN, RUNNING, PAUSED, STOPPED
            }
        }

        public interface ExperimentBuckets {
            Flux<Bucket> findAll();
            Flux<Bucket> subCollection(long from, long to);
            Mono<Bucket> findById(BucketId id);
        }
    }

            LocalDateTime now = LocalDateTime.now();

            Experiment experiment1 = Experiment.builder()
                .id(new ExperimentId("experiment1"))
                .description(new ExperimentDescription("name1", "description1", "condition1", 0.5, now, now, ExperimentStatus.DRAFT, DynamicStatus.DRAFT, "featureKey1", now, now))
                .build();

            experimentRepository.save(experiment1).block();

            Mono<Long> experimentCount = experimentRepository.count();

            StepVerifier.create(experimentCount)
                .expectSubscription()
                .expectNext(1L)
                .verifyComplete();
        }
    }
